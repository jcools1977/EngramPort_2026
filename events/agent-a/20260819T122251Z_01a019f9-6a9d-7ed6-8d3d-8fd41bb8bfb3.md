-- D1 behavioural guard assertions.
-- MUST run as engram_maintenance, never as postgres: postgres is a superuser and
-- bypasses RLS entirely, which makes the membership assertion meaningless.
\set ON_ERROR_STOP on
DO $$
DECLARE n integer; ns text; msg text;
BEGIN
  -- G1 membership_principal_self: no foreign membership row may be visible.
  PERFORM set_config('app.principal_id','11000000-0000-0000-0000-000000000001',false);
  PERFORM set_config('app.tenant_id','',false);

   WHERE principal_id<>'11000000-0000-0000-0000-000000000001';
  IF n<>0 THEN RAISE EXCEPTION 'G1 membership leak: % foreign row(s) visible', n; END IF;
  SELECT count(*) INTO n FROM project_memberships;
  IF n<>1 THEN RAISE EXCEPTION 'G1 own membership not visible: % row(s)', n; END IF;

  -- G2 custody-class FK. The tenant is set to the correct value so the RLS write
  -- policy permits the row; the FK must then be what refuses it, not RLS.
  PERFORM set_config('app.tenant_id','10000000-0000-0000-0000-000000000001',false);
  BEGIN
    INSERT INTO custody_rows(id,tenant_id,project_id,namespace,credential_class,
      custody_model,inventory_model,required_scope,minted_by_principal_id,retention_policy)
    VALUES(gen_random_uuid(),'10000000-0000-0000-0000-000000000001',
      '12000000-0000-0000-0000-000000000001','credential','G2-UNMAPPED','A','A','x',
      '11000000-0000-0000-0000-000000000001','RET-CONFIG-400');
    RAISE EXCEPTION 'G2 custody-class FK did not refuse an unmapped class';
  EXCEPTION
    WHEN foreign_key_violation THEN NULL;
    WHEN insufficient_privilege THEN RAISE EXCEPTION 'G2 masked: refused by RLS, not the FK';
  END;
  PERFORM set_config('app.tenant_id','',false);

  -- G3 M7 scope containment: an unheld scope must raise SCOPE_EXCEEDED specifically.
  BEGIN
    PERFORM mint_custody_reference('3.2','credential','B','vault:transit/g3','{}'::jsonb);
    RAISE EXCEPTION 'G3 scope containment did not refuse an unheld scope';

    IF msg<>'SCOPE_EXCEEDED' THEN RAISE EXCEPTION 'G3 refused with % not SCOPE_EXCEEDED', msg; END IF;
  END;

  -- G4 full M8. The seeded authority holds the shape and installation triples, so
  -- the namespace guard is the only control that can refuse. Any other refusal
  -- reason means a different guard masked M8.
  FOREACH ns IN ARRAY ARRAY['shape','installation'] LOOP

      PERFORM mint_custody_reference('3.12',ns::epr_namespace,'A',NULL,'{}'::jsonb);
      RAISE EXCEPTION 'G4 namespace % was not refused', ns;
    EXCEPTION WHEN insufficient_privilege THEN
      GET STACKED DIAGNOSTICS msg = MESSAGE_TEXT;
      IF msg<>'NAMESPACE_REFUSED' THEN RAISE EXCEPTION 'G4 masked: namespace % refused with %', ns, msg; END IF;
    END;
  END LOOP;

  RAISE NOTICE 'D1 behavioural guards OK';

### `scripts/run-d1-mutation-harness`

Replaces the placeholder. Implements the ten-step lifecycle: create isolated `engramport_mut`, apply canonical migrations, seed prerequisites, run the baseline as `engram_maintenance`, apply one verified mutation, observe the forbidden behaviour, require a nonzero run, rebuild, confirm the baseline passes, and drop the scratch database **under every exit path** via a `trap` on `EXIT INT TERM`.

```bash
#!/usr/bin/env bash
# D1 scratch-database mutation harness. Test-only: never edits migrations or
# production files. Each run builds an isolated engramport_mut, applies one
# mutation, verifies it applied, and requires the behavioural assertions to FAIL.
set -uo pipefail
root_dir="$(cd "$(dirname "$0")/.." && pwd)"
compose=(docker compose -f "$root_dir/deploy/docker-compose.yml")
SDB=engramport_mut
psql_as(){ "${compose[@]}" exec -T postgres psql -q -v ON_ERROR_STOP=1 -U "$1" -d "$2" "${@:3}"; }
drop_scratch(){ "${compose[@]}" exec -T postgres psql -q -U postgres -d postgres \
  -c "DROP DATABASE IF EXISTS $SDB WITH (FORCE);" >/dev/null 2>&1 || true; }
cleanup(){ rc=$?; drop_scratch; exit $rc; }
trap cleanup EXIT INT TERM

build(){
  drop_scratch
  psql_as postgres postgres -c "CREATE DATABASE $SDB OWNER engram_migrator;" >/dev/null || return 1
  psql_as postgres "$SDB" -c "CREATE EXTENSION IF NOT EXISTS vector; CREATE EXTENSION IF NOT EXISTS pgcrypto;" >/dev/null || return 1
  for m in "$root_dir"/migrations/000[1-9]_*.sql; do
    psql_as engram_migrator "$SDB" < "$m" >/dev/null || { echo "build failed: $m" >&2; return 1; }
  done
  psql_as postgres "$SDB" < "$root_dir/deploy/seed.sql" >/dev/null || return 1
  # G4 requires the shape and installation triples so the namespace guard is the
  # only control that can refuse; without them the scope check masks M8.
  psql_as postgres "$SDB" -c "INSERT INTO founder_authorities(principal_id,scopes,expires_at) VALUES ('11000000-0000-0000-0000-000000000001',ARRAY['custody:mint:credential:3.12:A','custody:mint:shape:3.12:A','custody:mint:installation:3.12:A'],clock_timestamp()+interval '2 hours') ON CONFLICT (principal_id) DO UPDATE SET scopes=EXCLUDED.scopes,expires_at=EXCLUDED.expires_at,revoked_at=NULL;" >/dev/null || return 1
}
assert(){ psql_as engram_maintenance "$SDB" -f - < "$root_dir/tests/failure/d1-behavioural.sql" >/dev/null 2>&1; echo $?; }

fail=0
# Read the whole list first: docker compose exec consumes stdin, which would
# otherwise swallow the remaining lines of a `while read` loop.
mapfile -t MUTATIONS < "$root_dir/tests/failure/d1-mutations.txt"
for line in "${MUTATIONS[@]}"; do
  [ -z "${line// }" ] && continue
  IFS='|' read -r name mutation check <<< "$line"
  build || { echo "FAIL $name: scratch build failed" >&2; fail=1; continue; }
  base=$(assert)
  psql_as postgres "$SDB" -c "$mutation" >/dev/null 2>&1
  applied=$(psql_as postgres "$SDB" -Atc "$check" | tr -d '[:space:]')
  after=$(assert)
  build >/dev/null 2>&1; restored=$(assert)
  printf '%-34s baseline=%s applied=%s after=%s restored=%s\n' "$name" "$base" "$applied" "$after" "$restored"
  [ "$base" = "0" ] && [ "$applied" = "t" ] && [ "$after" != "0" ] && [ "$restored" = "0" ] || fail=1
done

drop_scratch
[ "$fail" = "0" ] || { echo "D1 mutation harness: at least one control did not discriminate" >&2; exit 1; }
echo "D1 mutation harness: all controls discriminate"
```
